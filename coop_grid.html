coop_grid

Phase 12 grids and frozen analysis plans: Track A (instructions and channels for Claude: trackAGrid, trackAPilotGrid, TRACKA_PLAN) and Track B (RL_PLAN, evalSuite); coopHash. Phase 13, Track B: RL13_PLAN (arms and seeds over RL_PLAN v2), rl13Config, rl13Suite.

RL13_PLAN’s evaluation cells (evalSuite’s) for models (default: the five new adapters): all but cross-play, or only cross-play.

from explRL.coop_grid import *
from explRL.llm_grid import estimateGrid
from explRL.llm_runs import LLMConfig, cellRow
from explRL.bench_grid import rowKey

calls = lambda cells: sum(v['calls'] for v in estimateGrid(cells).values())
g, p = trackAGrid(), trackAPilotGrid()
assert len(g) == 4 * 2 * 5 and calls(g) == 9600 and len(p) == 4 and calls(p) == 480
assert 2.0 < sum(v['cost'] for v in estimateGrid(g).values()) < 2.6
keys = [rowKey(cellRow(c, LLMConfig.fromDict(c['cfg']), dict(episode=0))) for c in g + p]
assert len(set(keys)) == len(keys)
# options in the world, the Phase 11 board settings otherwise; A1 and A2 only on the coordinator
for c in g:
    w = c['cfg']['world']
    assert {k: w[k] for k in BOARD} == {k: CLUE_SETTINGS[k] for k in BOARD}
    assert (c['level'] in ('A1', 'A2')) <= (c['cfg']['channel'] == 'hierarchical')
assert {c['seed'] for c in g} == set(range(900, 905)) and all(c['cfg']['turns'] == 30 for c in p)
assert all(c['cfg']['turns'] == 3 for c in trackAQuick()) and len(trackAQuick()) == 4
# the evaluation suite: turns of local-model agents per model as in design-low-level section 8
turns = lambda cells, m: sum((c['N'] - c['cfg']['nLiar']) * c['cfg']['turns'] for c in cells if c['level'].startswith(m + '|'))
ev = evalSuite()
assert turns(ev, 'untrained') == 2400 and turns(ev, 'warm') == 2400 + 3200 + 2 * 800 + 720
assert sum(c['N'] * 40 for c in ev if c['level'].endswith(('cross1', 'cross3'))) == 4 * 2 * 5 * 160
assert all(c['runner'] == 'local' and c['cfg']['turns'] == 40 and c['cfg']['memory'] == 'notes' for c in ev)
assert len({rowKey(cellRow(c, LLMConfig.fromDict(c['cfg']), dict(episode=0))) for c in ev}) == len(ev)
assert len(evalSuite(extra=('perRole',))) == len(ev) + 15
assert RL_PLAN['version'] == 2 and RL_PLAN['rl']['turnFraction'] == 0.25 and RL_PLAN['rl']['gradCkpt'] is True
assert RL_PLAN['rl']['klTarget'] == 0.003 and RL_PLAN['rl']['lrMax'] >= RL_PLAN['rl']['lr'] and RL_PLAN['rl']['haltRun'] == 3
assert set(RL_PLAN['rl']) <= set(asdict(RLConfig('', '')))         # every setting is an RLConfig field
assert RL_PLAN['prices'] == TRACKA_PLAN['prices'] and RL_PLAN['rl']['steps'] == 60 and RL_PLAN['seeds']['main'] == MAIN_SEEDS and coopHash(RL_PLAN) != coopHash(TRACKA_PLAN)
assert len(coopHash(TRACKA_PLAN)) == 12 and coopHash({**TRACKA_PLAN, 'version': 2}) != coopHash(TRACKA_PLAN)
# NFR-1: RL_PLAN v2 as frozen (commit 304b955), without Phase 13's fields
assert coopHash(RL_PLAN) == '4d1ea0078a6e' and RL13_PLAN['base']['hash'] == '4d1ea0078a6e' and not set(RL13_DEFAULTS) & set(RL_PLAN['rl'])
# RL13_PLAN: five new runs; seed 1 of credit is Phase 12's v2 credit run (same identity); the arms differ from it only as listed
from explRL.clue_rl import configHash
W = '/content/drive/MyDrive/explRL_rl/warm_model'
cf = lambda a, s: asdict(rl13Config(a, s, W, 'o'))
assert configHash(rl13Config('credit', 1, W, 'o')) == 'c02fa4f72378' and len(RL13_RUNS) == 5
assert {k for k, v in cf('credit', 2).items() if v != cf('credit', 1)[k]} == {'trainSeed'}
assert {k: v for k, v in cf('indpen', 1).items() if v != cf('credit', 1)[k]} == dict(scheme='individual', wrongPenalty=2.5, haltJump=0.15)
assert {k: v for k, v in cf('credturn', 2).items() if v != cf('credit', 1)[k]} == dict(advantage='turn', wrongPenalty=2.5, haltJump=0.15, trainSeed=1)
assert len({configHash(rl13Config(a, s, W, 'o')) for a, s in RL13_RUNS + (('credit', 1),)}) == 6
assert [rl13Name(a, s) for a, s in RL13_RUNS] == ['credit_s2', 'indpen_s1', 'indpen_s2', 'credturn_s1', 'credturn_s2']
# the evaluation: main, generalisation and liar cells for the five new adapters on Colab; cross-play for two (the best arm) on the Mac
ev13 = rl13Suite()
assert len(ev13) == 5 * (15 + 15 + 6) and not any(c['level'].endswith(('cross1', 'cross3')) for c in ev13)
assert {c['level'].split('|')[0] for c in ev13} == {'credit_s2', 'indpen_s1', 'indpen_s2', 'credturn_s1', 'credturn_s2'}
cr = rl13Suite(['indpen_s1', 'indpen_s2'], cross=True)
assert len(cr) == 2 * 2 * 5 and all(c['level'].endswith(('cross1', 'cross3')) for c in cr)
assert RL13_PLAN['version'] == 1 and coopHash(RL13_PLAN) != coopHash(RL_PLAN)
