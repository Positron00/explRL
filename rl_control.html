rl_control

Phase 12, Track B: the pure control logic of the RL loop (no GPU libraries): the learning rate steered to a target per-step policy change (with warm-up, clamping and undoing of runaway steps), the stopping rule, micro-batching, and (Phase 13) the roll-back of an update after a one-step jump in invalid replies.

source

microBatches

def microBatches(
    samples, cfg
):

Chunks of at most microBatch samples; with per-role adapters every chunk holds one slot’s samples.

source

jumpGuard

def jumpGuard(
    play, jumped, undo, rolls:int=0
):

Roll-back on a one-step jump (Phase 13): play() -> the step’s rollouts; while jumped(rollouts) (with the last update kept, an invalid share above haltJump), that update is undone (undo()), the rollouts discarded and the step played again. rolls: roll-backs in a row before this step. -> (rollouts, roll-backs at this step, roll-backs in a row), with rollouts None after two in a row (the run halts). The caller halves the learning rate once per roll-back.

source

invalidShare

def invalidShare(
    rows
):

Invalid replies per text turn over a step’s board rows.

source

haltReason

def haltReason(
    history, cfg
):

The stopping rule: invalid replies above haltInvalid for haltRun steps in a row, or the mean KL above haltKL (haltKLRun steps in a row; default 1).

nextLr

def nextLr(
    lr, stepKL, cfg, warming:bool=False, trained:bool=True
):

The KL-targeted learning rate (off when cfg.klTarget is 0): -> (next learning rate, undo this step). A step whose policy change exceeds rejectFactor x klTarget is undone and the rate halved; above 2 x klTarget the rate is halved, below klTarget / 2 raised by half (not while warming up, when the step used only part of the rate, and not after a step with nothing to train on); always within [lrMin, lrMax].

source

warmLr

def warmLr(
    lr, s, warmup

The learning rate of step s (from 0): rises linearly to lr over the first warmup steps.

import math
from types import SimpleNamespace as C

m = dict(step=12, lr=2.2e-4, stepKL=0.0031, kl=0.041, welfare=0.2512, invalid=0.0, rejected=False, rolloutSeconds=60.2, updateSeconds=102.9)
assert stepLine(m, 60) == ('step 12/60: lr 0.00022, step KL 0.0031, KL to warm start 0.0410, welfare 0.251, invalid 0.000, '
                           'undone no, 163 s')
assert 'undone yes' in stepLine(dict(m, rejected=True), 60)
assert stepLine(dict(m, rolledBack=0), 60) == stepLine(m, 60) and stepLine(dict(m, rolledBack=1), 60).endswith('163 s, rolled back 1')
assert math.isnan(nanMean([float('nan')])) and nanMean([1.0, float('nan'), 3.0]) == 2.0
# warm-up: linear to lr over `warmup` steps, then lr
assert np.allclose([warmLr(1e-4, s, 4) for s in range(5)], [2.5e-5, 5e-5, 7.5e-5, 1e-4, 1e-4]) and warmLr(1e-4, 0, 0) == 1e-4
# the KL-targeted learning rate: steered to the per-step target, clamped, runaway steps undone; off without a target
c = C(klTarget=0.01, rejectFactor=4.0, lrMin=1e-6, lrMax=3e-4)
same = lambda a, b: math.isclose(a[0], b[0]) and a[1] == b[1]
assert same(nextLr(1e-4, 0.01, c), (1e-4, False))                    # in the band: unchanged
assert same(nextLr(1e-4, 0.025, c), (5e-5, False))                   # too far: halved
assert same(nextLr(1e-4, 0.002, c), (1.5e-4, False))                 # too little: raised by half
assert same(nextLr(1e-4, 0.05, c), (5e-5, True))                     # runaway: undone and halved
assert same(nextLr(2.5e-4, 0.001, c), (3e-4, False)) and same(nextLr(1.5e-6, 0.5, c), (1e-6, True))   # clamped
assert nextLr(1e-4, 9.9, C(klTarget=0.0)) == (1e-4, False)
# while warming up (the step used part of the rate) a small change does not raise it; too far still lowers or undoes
assert same(nextLr(1e-4, 0.002, c, warming=True), (1e-4, False)) and same(nextLr(1e-4, 0.05, c, warming=True), (5e-5, True))
# a step with nothing to train on (every group tied) leaves the rate alone
assert nextLr(1e-4, 0.0, c, trained=False) == (1e-4, False)
# the stopping rule: invalid replies too high for haltRun steps in a row, or the KL too high
h = lambda *xs: [dict(step=k + 1, invalid=v, kl=kl) for k, (v, kl) in enumerate(xs)]
c = C(haltRun=3, haltInvalid=0.10, haltKL=0.5)
assert haltReason(h((0.2, 0.0), (0.2, 0.0)), c) is None and haltReason(h((0.2, 0.0), (0.05, 0.0), (0.2, 0.0), (0.2, 0.0)), c) is None
assert 'steps 2-4' in haltReason(h((0.0, 0.0), (0.2, 0.0), (0.3, 0.0), (0.11, 0.0)), c)
assert 'KL 0.600' in haltReason(h((0.0, 0.1), (0.0, 0.6)), c) and haltReason([], c) is None
# Phase 18 Part B: with haltKLRun 2 a one-step KL outlier does not halt, two steps in a row do
c2 = C(haltRun=3, haltInvalid=0.10, haltKL=0.5, haltKLRun=2)
assert haltReason(h((0.0, 0.1), (0.0, 3.5)), c2) is None and haltReason(h((0.0, 3.5), (0.0, 0.02), (0.0, 0.9)), c2) is None
assert '2 steps in a row' in haltReason(h((0.0, 0.1), (0.0, 0.6), (0.0, 0.7)), c2)
# micro-batches: per-role chunks never mix slots
mb = microBatches([dict(slot=s) for s in (0, 1, 0, 1, 1)], C(perRole=True, microBatch=2))
assert [[x['slot'] for x in b] for b in mb] == [[0, 0], [1, 1], [1]]
assert [len(b) for b in microBatches([dict(slot=s) for s in (0, 1, 0, 1, 1)], C(perRole=False, microBatch=2))] == [2, 2, 1]
assert invalidShare([dict(invalid=1, calls=10), dict(invalid=2, calls=20)]) == 0.1 and invalidShare([]) == 0.0
# roll-back on a one-step jump (AC-4), with stub rollouts: each play() returns the next invalid share; undo() is counted
def guard(shares, rolls=0, limit=0.15):
    it, undone = iter(shares), []
    ro, back, rolls = jumpGuard(lambda: next(it), lambda x: x > limit, lambda: undone.append(1), rolls)
    return ro, back, rolls, len(undone), list(it)
assert guard([0.1, 0.9]) == (0.1, 0, 0, 0, [0.9])                    # no jump: played once
assert guard([0.5, 0.1, 0.9]) == (0.1, 1, 1, 1, [0.9])               # a jump: undone, discarded, the step played again
assert guard([0.1], rolls=1) == (0.1, 0, 0, 0, [])                   # a step without a roll-back ends the run of them
assert guard([0.5, 0.6, 0.1]) == (None, 2, 2, 2, [0.1])              # the redone step jumps again: halt
assert guard([0.5, 0.1], rolls=1) == (None, 1, 2, 1, [0.1])          # a roll-back right after one at the step before: halt
assert guard([0.9, 0.9], limit=1.0) == (0.9, 0, 0, 0, [0.9])         # haltJump off (never jumped): Phase 12 behaviour
