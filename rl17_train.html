rl17_train

Phase 17: trainRL17, trainRL‘s GRPO loop (its step pieces in rl17_update) with three changes: each step’s boards are split across team sizes (mixN) with scripted liars on a share of them (liarShare); the reference and old log-probabilities are computed once per step and reused by epochs update passes with a clipped reply-level ratio (clip); replies can be sampled under the reply schema (guided), and lrFixed switches the per-step controller off; pilot v3’s fix (valueOnly, tokenClip, a longer maxLen) and a per-step count of cut prompts (cut) and the longest prompt plus reply (longest); Phase 18’s world (the board options of every rollout, and the faulty slots’ class). At the defaults one step equals trainRL’s apart from the added cut and longest (tested on the tiny model). Training needs the [train] extra; the offline tests do not.

Tests

import numpy as np
from explRL.rl17_train import *
from explRL.rl17_grid import RL17Config
# With the [train] extra, no download: at the defaults trainRL17 equals trainRL (metrics and final adapter) on the tiny model;
# a mixed step with liars and clipped passes runs; its checkpoint carries configHash17
import importlib.util
if all(importlib.util.find_spec(m) for m in ('transformers', 'peft', 'tokenizers')):   # the [train] extra; the docs build runs cells with imports and lacks it
    import json
    import tempfile
    import torch
    from dataclasses import asdict
    from explRL.clue_rl import RLConfig, trainRL
    import explRL.rl17_train as T
    from explRL.tiny_model import tinyModel
    from safetensors.torch import load_file
    _tr = T.trainingReturns
    with tempfile.TemporaryDirectory() as d:
        tiny = tinyModel(d + '/tiny')
        kw = dict(steps=2, seedsPerStep=2, group=3, turns=3, ckptEvery=1, maxNew=24, lr=1e-3, haltRun=99)
        import explRL.clue_rl as C
        for mod in (C, T):                       # the tiny model's replies are invalid: the same random returns stand in for both
            rng = np.random.default_rng(0)
            mod.trainingReturns = (lambda r: (lambda ro, *a: _tr(ro, *a) + r.normal(size=np.shape(ro.returns))))(rng)
            if mod is C:
                trainRL(RLConfig(tiny, f'{d}/a', **kw))
            else:
                trainRL17(RL17Config(tiny, f'{d}/b', **kw))
        strip = lambda p: [{k: v for k, v in json.loads(x).items() if 'Seconds' not in k and k not in ('cut', 'longest')} for x in open(p)]
        assert strip(f'{d}/a/metrics.jsonl') == strip(f'{d}/b/metrics.jsonl')
        wa, wb = load_file(f'{d}/a/final/adapter_model.safetensors'), load_file(f'{d}/b/final/adapter_model.safetensors')
        assert wa.keys() == wb.keys() and all(torch.equal(wa[k], wb[k]) for k in wa)
        T.trainingReturns = (lambda r: (lambda ro, *a: _tr(ro, *a) + r.normal(size=np.shape(ro.returns))))(np.random.default_rng(1))
        cm = RL17Config(tiny, f'{d}/m', **{**kw, 'seedsPerStep': 4}, mixN=((4, 0.5), (8, 0.5)), liarShare=0.5, epochs=2, clip=0.2, kl=0.0, lrFixed=1e-3,
                        valueOnly=True, tokenClip=True, maxLen=4096)
        trainRL17(cm)
        m = [json.loads(x) for x in open(f'{d}/m/metrics.jsonl')]
        assert [x['step'] for x in m] == [1, 2] and all(x['samples'] > 0 and np.isfinite(x['loss']) for x in m)
        assert all(x['cut'] == 0 and np.isfinite(x['klAll']) and 0 < x['longest'] <= 4096 for x in m)   # pilot v3's fix runs; nothing cut at 4096
        # Phase 18 Part B's fix: every token's k3 capped, so the logged KL (value tokens and all) can never exceed the cap
        ck = RL17Config(tiny, f'{d}/k', **{**kw, 'lr': 5e-2}, kl=0.04, lrFixed=5e-2, valueOnly=True, tokenClip=True, maxLen=4096,
                        klClamp=1e-4, haltKLRun=2)
        trainRL17(ck)
        mk = [json.loads(x) for x in open(f'{d}/k/metrics.jsonl')]
        assert all(x['kl'] <= 1e-4 + 1e-12 and x['klAll'] <= 1e-4 + 1e-12 for x in mk), mk
        # Phase 18: a step on the colony board (two holders, colluding liars) through `world`
        cw = RL17Config(tiny, f'{d}/w', **{**kw, 'steps': 1, 'seedsPerStep': 2}, mixN=((8, 1.0),), liarShare=0.5, maxLen=4096,
                        world=(('colony', True), ('r', 2), ('fault', 'collude')))
        trainRL17(cw)
        assert json.load(open(f'{d}/w/config.json'))['hash'] == T.configHash17(cw) != T.configHash17(RL17Config(**{**asdict(cw), 'world': ()}))
        recs = [json.loads(x) for x in open(f'{d}/w/turns/step-0001.jsonl')]
        assert recs and all(r['N'] == 8 and 'Clues shared on the board:' in r['obs'] and '(agent ' not in r['obs'] for r in recs)
        assert {r['nLiar'] for r in recs} == {0, 2}
        assert json.load(open(f'{d}/m/config.json'))['hash'] == T.configHash17(cm)
        T.trainingReturns = _tr
        C.trainingReturns = _tr
