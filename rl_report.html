rl_report

Phase 12, Track B report: the RL_PLAN analyses of the evaluation rows (main boards, gains over the warm-start model, reward schemes, Phase 11 Haiku and rule agents, information use, generalisation, follow share, cross-play, liar, per-role adapters), report.md with provenance first, and figures. Phase 13: adapters named <arm>_s<seed> are grouped by arm (per-arm mean of the per-seed RL gains, each seed’s gain) and the best arm picked. All findings are exploratory.

bestArm

def bestArm(
    S, arms:NoneType=None
):

RL13_PLAN’s rule: the arm with the highest mean RL gain over the warm start on the main boards (among arms).

source

rlSummaries

def rlSummaries(
    rows, p11, numpyRows:tuple=()
):

RL_PLAN: main, gains, schemes, against Phase 11, information, generalisation, follow share, cross-play, liar, per-role; RL13_PLAN: per arm (adapters ’_s‘), the mean of its seeds’ RL gains with a CI from the pooled differences.

import tempfile
from pathlib import Path

# synthetic evaluation rows with known differences (welfare = base + model offset, paired on seed)
OFF = dict(untrained=0.0, warm=0.2, individual=0.25, team=0.3, credit=0.4)
def row(model, test, ch, N, seed, w, **kw):
    r = dict(study='coopEval', level=f'{model}|{test}', channel=ch, N=N, seed=seed, episode=0, runner='local', welfare=w,
             welfareHonest=w, pooledShare=w / 2, followShare=0.5, **{k: 1.0 for k in INFO})
    r.update(kw)
    return r
base = lambda s: 0.1 + 0.01 * (s % 5)
rows = [row(m, 'main', 'blackboard', 4, s, base(s) + o) for m, o in OFF.items() for s in list(range(900, 905)) + list(range(930, 940))]
for m, o in OFF.items():
    if m == 'untrained':
        continue
    rows += [row(m, t, ch, N, s, base(s) + o) for t, ch, N in (('gen8', 'blackboard', 8), ('genMsg', 'messages', 4),
                                                            ('genLead', 'hierarchical', 4), ('cross1', 'blackboard', 4),
                                                            ('cross3', 'blackboard', 4)) for s in range(900, 905)]
    rows += [row(m, 'liar', ch, 4, s, base(s) + o - (0.1 if ch == 'blackboard' else 0.05)) for ch in ('blackboard', 'messages') for s in range(900, 903)]
p11 = [dict(study='clue', runner='llm', channel=ch, N=N, seed=s, episode=0, welfare=base(s) + 0.3)
       for ch, N in (('blackboard', 4), ('blackboard', 8), ('messages', 4), ('hierarchical', 4)) for s in range(900, 905)]
rule = [dict(study='clueNumpy', runner='numpy', channel='blackboard', N=4, seed=s, episode=0, welfare=base(s) + 0.5) for s in range(900, 905)]
S = rlSummaries(rows, p11, rule)
near = lambda v, x: abs(v['mean'] - x) < 1e-9
assert list(S['main']) == list(OFF) and S['main']['credit']['welfare']['n'] == 15
assert near(S['gain']['credit'], 0.2) and near(S['gain']['individual'], 0.05) and 'warm' not in S['gain']
assert near(S['schemes']['credit - individual'], 0.15) and near(S['schemes']['credit - team'], 0.1)
assert near(S['phase11']['credit - Haiku'], 0.1) and near(S['phase11']['credit - rule agents'], -0.1) and S['phase11']['credit - Haiku']['n'] == 5
assert near(S['gen']['warm gen8 - Haiku'], -0.1) and near(S['cross']['team cross1 - all Haiku'], 0.0) and near(S['follow']['credit'], 0.5)
assert near(S['liar']['credit honest loss, blackboard'], 0.1) and near(S['liar']['credit honest loss, messages'], 0.05)
assert near(S['liar']['credit blackboard - messages loss (H3: > 0)'], 0.05) and not S['perRole']
with tempfile.TemporaryDirectory() as d:
    meta = dict(mode='full', planHash='p', commit='c', base='m', adapters=dict(credit='abcdef0123456789'), gpuHours=5.5,
                gpuCost=12.0, apiCost=0.8, apiSpent=48.0, halted=dict(team=dict(step=7, reason='mean KL 0.612 above 0.5 at step 7')))
    text = writeRL(Path(d) / 'report.md', S, meta).read_text()
    assert '**Run team stopped by the stopping rule at step 7:** mean KL 0.612' in text
    assert text.index('`p`') < text.index('## Main boards') and '| credit - individual |' in text and 'Per-role' not in text
    curves = {sc: [dict(step=k, welfare=0.1 * k, sharePerHeld=0.2) for k in (1, 2, 3)] for sc in SCHEMES}
    paths = writeRLFigures(Path(d), S, curves, refs=[(0.4, 'Haiku', 'tab:orange')])
    assert [p.name for p in paths] == ['coop_main.png', 'coop_info.png', 'coop_gen.png', 'coop_curves.png']
assert not S['arms'] and 'per arm' not in text and armSeed('credit') is None and armSeed('perRole') is None
# Phase 13: adapters '<arm>_s<seed>' grouped by arm; Phase 12's credit run relabelled as seed 1 of the credit arm
OFF13 = dict(credit_s2=0.3, indpen_s1=0.1, indpen_s2=0.3, credturn_s1=0.35, credturn_s2=0.45)
r13 = renameModel([r for r in rows if r['level'].split('|')[0] in ('warm', 'credit')], 'credit', 'credit_s1')
r13 += [row(m, 'main', 'blackboard', 4, s, base(s) + o) for m, o in OFF13.items() for s in list(range(900, 905)) + list(range(930, 940))]
assert {r['level'] for r in r13 if r['level'].startswith('credit_s1')} == {f'credit_s1|{t}' for t in ('main', 'gen8', 'genMsg', 'genLead', 'cross1', 'cross3', 'liar')}
S13 = rlSummaries(r13, p11, rule)
assert list(S13['main']) == ['warm', 'credit_s1', 'credit_s2', 'credturn_s1', 'credturn_s2', 'indpen_s1', 'indpen_s2']
assert near(S13['gain']['credturn_s1'], 0.15) and near(S13['gain']['indpen_s1'], -0.1) and near(S13['gain']['credit_s1'], 0.2)
assert list(S13['arms']) == ['credit', 'credturn', 'indpen'] and near(S13['arms']['credturn'], 0.2) and S13['arms']['credturn']['n'] == 30
assert np.allclose(list(S13['arms']['credit']['seeds'].values()), [0.2, 0.1]) and near(S13['arms']['indpen'], 0.0)
assert bestArm(S13) == 'credturn' and bestArm(S13, ('credit', 'indpen')) == 'credit'
with tempfile.TemporaryDirectory() as d:
    text = writeRL(Path(d) / 'report.md', S13, dict(meta, title='Phase 13, Track B: RL that beats imitation', plan='RL13_PLAN')).read_text()
    assert text.startswith('# Phase 13, Track B: RL that beats imitation (full)') and 'pre-specified in `RL13_PLAN`' in text
    assert '## RL gain per arm' in text and '| credturn | 0.200 [' in text
